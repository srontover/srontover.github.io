\section{Research Experience}\color{black}
\cvproject{Learning Dynamic Skills for Humanoid Robots via Reinforcement Learning}{2026--Present}{T1 Humanoid Robot Kicking and Locomotion}
\begin{itemize}
  \item Implemented and modified reinforcement-learning environments for humanoid locomotion and contact-rich kicking tasks using MuJoCo/MJLab/Warp simulation workflows.
  \item Implemented and evaluated task components including action and observation spaces, ball reset logic, contact detection, reward functions, termination conditions, and curriculum learning procedures.
  \item Conducted experiments to analyze policy behavior through contact quality, ball velocity, target direction error, post-kick recovery, and training-stability metrics.
  \item Investigating Conditional AMP and motion-prior-based kicking policies for skill generalization across different ball positions and kicking directions.
\end{itemize}

\cvproject{Conditional AMP for Generalizable Humanoid Kicking}{2026}{RL Research Project}
\begin{itemize}
  \item Investigated distance-conditioned Conditional AMP for humanoid kicking using condition vectors over target distance and kick type with curriculum learning.
  \item Studying how expert-policy condition alignment and motion priors influence training stability and cross-condition skill generalization.
\end{itemize}

\cvproject{Motion Prior Learning for Humanoid Skill Acquisition}{2026}{Research Exploration}
\begin{itemize}
  \item Studying AMP-style motion priors, human motion data, encoder-decoder policies, and latent representations for humanoid skill learning.
  \item Exploring methods for balancing task rewards and motion-prior rewards to achieve natural and effective robot behaviors.
\end{itemize}

\cvproject{Robot Hand-Eye Calibration with 3D Sensing and Geometric Constraints}{2026}{Research Project | Manuscript in Preparation | Nanjing University of Posts and Telecommunications}
\begin{itemize}
  \item Conducting research on robot hand-eye calibration with a completed patent application and ongoing manuscript preparation.
  \item Investigating multi-stage calibration methods based on orthogonal geometric constraints, point-cloud filtering, and optimization.
  \item Conducted calibration experiments with ABB industrial robots and 3D sensing systems for evaluation.
\end{itemize}
